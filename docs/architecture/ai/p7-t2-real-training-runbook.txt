## Cell 2 - extract the private bundle into the working directory

```python
from pathlib import Path
import zipfile

working = Path("/kaggle/working").resolve()
matches = sorted(Path("/kaggle/input").rglob("p7-t2-real-training.zip"))
assert len(matches) == 1, f"Expected one uploaded bundle ZIP, found: {matches}"
with zipfile.ZipFile(matches[0]) as archive:
    for member in archive.infolist():
        target = (working / member.filename).resolve()
        assert target == working or working in target.parents, member.filename
    archive.extractall(working)
print(matches[0])
```

## Cell 7 - execute canonical real P7-T2 training

```python
from pathlib import Path
import json
import subprocess

bundle = json.loads(Path("bundle-manifest.json").read_text(encoding="utf-8"))
output = Path("/kaggle/working/p7-t2-real-training-output")
assert not output.exists(), output
command = [
    "python", "scripts/training-pipeline-p7-t2.py",
    "--config", "config/p7-t2-training-pipeline.json",
    "--decisions", "config/p6-t6-adapter-decisions.json",
    "--dataset-manifest", "datasets/p7-research-synthetic-training-dataset-v1/manifest.json",
    "--model-path", "/kaggle/working/p7-t2-model/cdbee75f17c01a7cc42f958dc650907174af0554",
    "--output", str(output),
    "--source-commit", bundle["sourceCommit"],
]
assert "--smoke" not in command
subprocess.run(command, check=True)
```

The configured run performs 1000 optimizer steps over only `train.jsonl`.
`validation.jsonl` is evaluated after training and is not optimization input.

## Cell 9 - create the return archive

```python
from pathlib import Path
import zipfile

output = Path("/kaggle/working/p7-t2-real-training-output")
archive_path = Path("/kaggle/working/p7-t2-real-training-output.zip")
assert output.is_dir() and not archive_path.exists()
with zipfile.ZipFile(archive_path, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
    for path in sorted(item for item in output.rglob("*") if item.is_file()):
        info = zipfile.ZipInfo(
            f"p7-t2-real-training-output/{path.relative_to(output).as_posix()}",
            date_time=(1980, 1, 1, 0, 0, 0),
        )
        info.compress_type = zipfile.ZIP_DEFLATED
        info.external_attr = 0o100644 << 16
        archive.writestr(info, path.read_bytes(), compress_type=zipfile.ZIP_DEFLATED, compresslevel=9)
print(archive_path)
```

Download `p7-t2-real-training-output.zip`. Do not upload the downloaded base
model or Hugging Face cache.
